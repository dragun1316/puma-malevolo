from PIL import Image, ImageDraw, ImageFilter import numpy as np, math, os src_path = "/mnt/data/3af17cbd-90e3-41cf-9837-0f5bef6f2089.jpg" im = Image.open(src_path).convert("RGBA") W, H = im.size # Isolate the puma and flower hat from the original poster with a soft elliptical mask. # The subject is near the upper center of the supplied image. x0, y0, x1, y1 = 225, 38, 532, 315 crop = im.crop((x0, y0, x1, y1)) cw, ch = crop.size mask = Image.new("L", (cw, ch), 0) md = ImageDraw.Draw(mask) # Broad oval around hat/head, feathered so the background blends smoothly. md.ellipse((5, 2, cw-5, ch-2), fill=255) mask = mask.filter(ImageFilter.GaussianBlur(9)) crop.putalpha(mask) frames = [] n = 24 for i in range(n): # Two exaggerated laugh bobs, with a small tilt and squash/stretch. phase = 2 * math.pi * i / n bob = int(4 * math.sin(phase * 2)) angle = 2.2 * math.sin(phase * 2) scale = 1.0 + 0.025 * (0.5 + 0.5 * math.sin(phase * 2)) nw, nh = int(cw*scale), int(ch*scale) subject = crop.resize((nw, nh), Image.Resampling.BICUBIC) subject = subject.rotate(angle, resample=Image.Resampling.BICUBIC, expand=True) frame = im.copy() px = (W - subject.width)//2 py = y0 + bob - (subject.height-ch)//2 frame.alpha_composite(subject, (px, py)) frames.append(frame.convert("RGB")) out_path = "/mnt/data/puma_risa_malevola.gif" frames[0].save(out_path, save_all=True, append_images=frames[1:], duration=75, loop=0, optimize=True) print(f"Animación GIF creada: {out_path} ({W}x{H}, {n} fotogramas)")
